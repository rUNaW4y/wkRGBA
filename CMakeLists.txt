cmake_minimum_required(VERSION 3.21)
project(wkRGBA VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "wkRGBA requires Windows x86: configure with -A Win32")
endif()
add_library(rgba_core STATIC src/Terrain.cpp src/Png.cpp src/D3D9Terrain.cpp src/Transfer.cpp src/ReplayData.cpp)
target_include_directories(rgba_core PUBLIC src)
target_compile_definitions(rgba_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(rgba_core PUBLIC windowscodecs ole32 d3d9 bcrypt)
add_library(wkRGBA SHARED src/Module.cpp src/Loader.cpp src/RenderProbe.cpp src/ExperimentalScene.cpp src/NativeCompositor.cpp src/WormLabels.cpp src/ReplayHooks.cpp src/NetworkHooks.cpp src/wkRGBA.def)
target_link_libraries(wkRGBA PRIVATE rgba_core)
add_executable(wkRGBATool src/Tool.cpp)
target_link_libraries(wkRGBATool PRIVATE rgba_core)
foreach(target rgba_core wkRGBA wkRGBATool)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(rgba_tests tests/Tests.cpp)
  target_link_libraries(rgba_tests PRIVATE rgba_core wkRGBA)
  target_compile_options(rgba_tests PRIVATE /W4 /permissive- /utf-8)
  set_property(TARGET rgba_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_test(NAME rgba_tests COMMAND rgba_tests)
  add_executable(transfer_tests tests/TransferTests.cpp)
  target_link_libraries(transfer_tests PRIVATE rgba_core)
  target_compile_options(transfer_tests PRIVATE /W4 /permissive- /utf-8)
  set_property(TARGET transfer_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_test(NAME transfer_tests COMMAND transfer_tests)
  add_executable(loader_tests tests/LoaderTests.cpp src/Loader.cpp src/RenderProbe.cpp src/ExperimentalScene.cpp src/NativeCompositor.cpp src/WormLabels.cpp src/ReplayHooks.cpp src/NetworkHooks.cpp)
  target_compile_definitions(loader_tests PRIVATE WKRGBA_LOADER_TESTING)
  target_compile_options(loader_tests PRIVATE /W4 /permissive- /utf-8)
  target_link_libraries(loader_tests PRIVATE rgba_core)
  set_property(TARGET loader_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_test(NAME loader_tests COMMAND loader_tests)
  add_executable(replay_tests tests/ReplayTests.cpp)
  target_link_libraries(replay_tests PRIVATE rgba_core)
  target_compile_options(replay_tests PRIVATE /W4 /permissive- /utf-8)
  set_property(TARGET replay_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_test(NAME replay_tests COMMAND replay_tests)
endif()
